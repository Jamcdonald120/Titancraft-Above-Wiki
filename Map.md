// Canvas dimensions const imgWidth = 3200; const imgHeight = 3200; const imageBounds = \[\[0, 0\], \[imgHeight, imgWidth\]\]; const map = L.map('map', { crs: L.CRS.Simple, minZoom: -2, maxZoom: 2, zoomSnap: 0.5 }); // Load the base map image L.imageOverlay('the\_above\_map.png', imageBounds).addTo(map); map.fitBounds(imageBounds); // Define explicit pin icon with bottom-center anchor point \[12, 41\] const defaultIcon = L.icon({ iconUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-icon.png', shadowUrl: 'https://unpkg.com/leaflet@1.9.4/dist/images/marker-shadow.png', iconSize: \[25, 41\], iconAnchor: \[12, 41\], popupAnchor: \[1, -34\] }); // Master layer group containing all searchable markers const allMarkersLayer = L.layerGroup().addTo(map); // Store category layer groups dynamically for toggle controls const categoryLayers = {}; // Fetch the markers fetch('minecraft\_markers.json') .then(res => res.json()) .then(markers => { markers.forEach(poi => { // Convert GIMP Top-Left Y to Leaflet Bottom-Left Y coordinate space const leafletY = imgHeight - poi.y; const leafletX = poi.x; const category = poi.group || "General"; // Create category layer group if it doesn't exist if (!categoryLayers\[category\]) { categoryLayers\[category\] = L.layerGroup().addTo(map); } // Build optional HTML strings for Event Type and Players const eventTypeHTML = poi.event\_type ? \`<p><b>Type:</b> ${poi.event\_type}</p>\` : ''; const playersHTML = poi.players ? \`<p><b>Players:</b> ${poi.players}</p>\` : ''; // Create individual marker const marker = L.marker(\[leafletY, leafletX\], { icon: defaultIcon, title: poi.name }).bindPopup(\` <div class="poi-popup"> <h3>${poi.name}</h3> <p><b>Group:</b> ${category}</p> ${eventTypeHTML} ${playersHTML} </div> \`); // Add marker to category group and master search group marker.addTo(categoryLayers\[category\]); marker.addTo(allMarkersLayer); }); // Add category filter toggles L.control.layers(null, categoryLayers, { collapsed: false }).addTo(map); // Add Search Control const searchControl = new L.Control.Search({ layer: allMarkersLayer, propertyName: 'title', initial: false, zoom: 1, marker: false, moveToLocation: function(latlng, title, map) { map.setView(latlng, 1); allMarkersLayer.eachLayer(function(marker) { if (marker.options.title === title) { marker.openPopup(); } }); } }); map.addControl(searchControl); });
